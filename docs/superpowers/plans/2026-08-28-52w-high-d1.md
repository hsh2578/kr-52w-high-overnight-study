# 52주 신고가 돌파 D+1 예측 연구 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 52주 신고가를 돌파한 종목의 익일 시가청산 수익률을 예측하고, 피처군(기존 14 / D+1 특화 10 / qlib Alpha158 계열 20 / 재료 5)의 기여도를 5단 ablation으로 측정해 학회 발표용 리포트와 그림을 산출한다.

**Architecture:** 순수 함수 중심의 파이프라인. `sources`(로컬 캐시 읽기) → `events`(이벤트+라벨) → `features_f1~f4`(피처군별 독립 모듈) → `dataset`(조립 후 parquet) → `validate`(walk-forward·embargo·가중·보정·통계검정) → `models`(M0~M4) → `report`(figure+리포트). 각 피처 모듈은 `(df, i, ctx) -> dict` 시그니처를 공유해 서로 독립적으로 테스트된다. 네트워크는 `sources`와 `predict.py`에만 존재한다.

**Tech Stack:** Python 3.13, pandas 2.3, numpy, LightGBM 4.6, scikit-learn 1.8, scipy 1.17, matplotlib 3.10, FinanceDataReader 0.9.101, pytest

**Spec:** `docs/superpowers/specs/2026-08-28-52w-high-d1-design.md`

## Global Constraints

- 왕복거래비용은 **0.0021** 고정 (상수 `COST`). 스펙 §6.
- 이벤트 조건: `Close[i] > max(Close[i-252:i])` **AND** `Volume[i] >= 1.5 * nanmean(Volume[i-50:i])` **AND** `Close[i]*Volume[i] >= 1e10` **AND** `i >= 260` **AND** `i+1 < len(df)` **AND** 최근 5봉과 익일 봉에 거래정지가 없을 것. 스펙 §3.1.
- **거래정지 봉 처리 (필수)**: 원본 데이터는 거래정지일을 `Open=0, Volume=0, Close=직전값`으로

```python
def _df(dates, o, h, l, c, v):
    return pd.DataFrame(
        {"Open": o, "High": h, "Low": l, "Close": c, "Volume": v},
        index=pd.to_datetime(dates),

def test_build_market_index_등가중수익률과_갭을_계산한다():
    d = ["2024-01-02", "2024-01-03"]

        "000001": _df(d, [100, 110], [110, 120], [95, 105], [100, 120], [10, 10]),
```

`surge_d1/sources.py`:

```python
# -*- coding: utf-8 -*-
"""데이터 로더. 로컬 캐시만 읽고 네트워크를 쓰지 않는다.

pickle 사용 주의: 읽는 pkl은 1st-party 로컬 캐시다(HollyKR 프로젝트가 자기
머신에서 쓴 것, 또는 scripts/refresh_cache.py 가 쓴 것). 외부 파일을
역직렬화하지 않는다.
"""
from __future__ import annotations

import pickle
import re

import numpy as np
import pandas as pd

from surge_d1 import config

OHLCV_COLS = ["Open", "High", "Low", "Close", "Volume"]


def mark_halted(df: pd.DataFrame) -> pd.DataFrame:
    """거래정지 봉을 표시하고 Open/Volume을 NaN으로 바꾼다.

    원본(FDR/네이버)은 거래정지일을 Open=0·Volume=0·Close=직전값으로 채운다.
    예: 카카오(035720) 2021-04-15 액면분할 처리로 04-12~14 정지.
    이 값을 그대로 두면 라벨이 −100%가 되고 거래량 평균이 왜곡되므로 NaN 처리한다.
    Close는 직전값이 그대로 유효한 참조가이므로 건드리지 않는다.
    """
    out = df.copy()
    halted = (out["Open"].fillna(0) <= 0) | (out["Volume"].fillna(0) <= 0)
    out["halted"] = halted.to_numpy()
    out.loc[halted, "Open"] = np.nan
    out.loc[halted, "Volume"] = np.nan
    return out


def load_ohlcv() -> dict[str, pd.DataFrame]:
    """갱신본이 있으면 그것을, 없으면 HollyKR 원본을 읽는다.

    원본은 key가 "{code}_{days}" 형태라 code별 최장 시계열만 남긴다.
    반환 전에 mark_halted를 적용한다.
    """
    if config.CACHE_PATH.exists():
        with open(config.CACHE_PATH, "rb") as f:
            cache = pickle.load(f)
        if cache and not re.match(r"\d+_\d+$", next(iter(cache))):
            return {k: mark_halted(v[OHLCV_COLS])
                    for k, v in cache.items() if v is not None}
    else:
        path = config.HOLLYKR_CACHE / "ohlcv" / "ohlcv_cache.pkl"
        with open(path, "rb") as f:
            cache = pickle.load(f)

    best: dict[str, pd.DataFrame] = {}
    for key, df in cache.items():
        m = re.match(r"(\d+)_(\d+)$", str(key))
        code = m.group(1) if m else str(key)
        if df is None or len(df) == 0:
            continue
        if code not in best or len(df) > len(best[code]):
            best[code] = df
    return {c: mark_halted(d[OHLCV_COLS]) for c, d in best.items()}


def load_catalyst() -> pd.DataFrame:
    """재료 CSV → 정규화된 DataFrame.

    반환 컬럼: date, code, ctype, reason, day_chg, d1
    """
    raw = pd.read_csv(config.CATALYST_CSV, encoding="utf-8")
    raw.columns = ["date", "name", "code", "chg_s", "day_chg",
                   "ctype", "reason", "d1", "d2", "d3", "cum3"]
    out = pd.DataFrame({
        "date": pd.to_datetime(raw["date"]),
        "code": raw["code"].astype(str).str.zfill(6),
        "ctype": raw["ctype"].fillna("").astype(str).str.strip(),
        "reason": raw["reason"].fillna("").astype(str).str.strip(),
        "day_chg": pd.to_numeric(raw["day_chg"], errors="coerce"),
        "d1": pd.to_numeric(raw["d1"], errors="coerce"),
    })
    return out.reset_index(drop=True)


def build_market_index(ohlcv: dict[str, pd.DataFrame]) -> pd.DataFrame:
    """전 종목 등가중 시장 지수. FDR의 KS11 호출이 불안정해 대용으로 쓴다.

    반환: index=Date, columns=[ret, gap, n]
      ret = 종목별 (Close/Close[-1] - 1) 의 그날 평균
      gap = 종목별 (Open/Close[-1] - 1) 의 그날 평균
      n   = 그날 계산에 쓰인 종목 수
    """
    rets, gaps = [], []
    for df in ohlcv.values():
        c = df["Close"].astype(float)
        o = df["Open"].astype(float)
        prev = c.shift(1)
        rets.append((c / prev - 1.0).rename(None))
        gaps.append((o / prev - 1.0).rename(None))
    r = pd.concat(rets, axis=1)
    g = pd.concat(gaps, axis=1)
    out = pd.DataFrame({
        "ret": r.mean(axis=1, skipna=True),
        "gap": g.mean(axis=1, skipna=True),
        "n": r.count(axis=1).astype(int),
    }).sort_index()
    out.loc[out["n"] == 0, ["ret", "gap"]] = np.nan
    return out
```

```python
# -*- coding: utf-8 -*-
"""52주 신고가 돌파 이벤트 스크린과 D+1 라벨. 스펙 §3.1, §6."""
from __future__ import annotations

import numpy as np
import pandas as pd

from surge_d1 import config


def is_breakout(df: pd.DataFrame, i: int) -> bool:
    """봉 i가 52주 신고가 돌파 이벤트인가.

    조건 (스펙 §3.1): 신고가 · 거래량 1.5배 · 거래대금 100억 · i>=260 · 익일 존재
    · 최근 HALT_LOOKBACK봉과 익일 봉에 거래정지 없음
    """
    if i < config.MIN_BAR_INDEX or i + 1 >= len(df):
        return False

    # 거래정지 배제: 정지 구간은 Open/Volume이 NaN이라 지표가 왜곡된다
    if "halted" in df.columns:
        lo = max(0, i - config.HALT_LOOKBACK + 1)
        if bool(df["halted"].iloc[lo:i + 2].any()):
            return False

    c = float(df["Close"].iloc[i])
    v = float(df["Volume"].iloc[i])
    if not np.isfinite(c) or not np.isfinite(v) or c <= 0:
        return False

    prior_high = float(df["Close"].iloc[i - config.HIGH_52W_PERIOD:i].max())
    if not (c > prior_high):
        return False

    # NaN(정지봉)을 건너뛴 평균
    vol_avg = float(df["Volume"].iloc[i - 50:i].mean())
    if not (np.isfinite(vol_avg) and vol_avg > 0
            and v >= config.VOLUME_SURGE_MULT * vol_avg):
        return False

    return bool(c * v >= config.MIN_TRADING_VALUE)


def labels(df: pd.DataFrame, i: int, cost: float = config.COST) -> dict:
    """봉 i 종가 매수 기준 라벨 3종. 스펙 §6.

    y_raw  = (Open[i+1]-Close[i])/Close[i] - cost   (익일 시가청산)
    y_hold = (Close[i+1]-Close[i])/Close[i] - cost  (익일 종가청산)
    y_cls  = 1 if y_raw > 0 else 0
    """
    c = float(df["Close"].iloc[i])
    o1 = float(df["Open"].iloc[i + 1])
    c1 = float(df["Close"].iloc[i + 1])
    y_raw = (o1 - c) / c - cost if (c > 0 and o1 > 0) else np.nan
    y_hold = (c1 - c) / c - cost if (c > 0 and c1 > 0) else np.nan
    return {
        "y_raw": y_raw,
        "y_hold": y_hold,
        "y_cls": int(y_raw > 0) if np.isfinite(y_raw) else 0,
    }


def collect_events(ohlcv: dict[str, pd.DataFrame]) -> pd.DataFrame:
    """전 종목 순회하며 이벤트와 라벨을 모은다."""
    rows = []
    for code, df in ohlcv.items():
        n = len(df)
        if n < config.MIN_BAR_INDEX + 2:
            continue
        c = df["Close"].to_numpy(dtype=float)
        v = df["Volume"].to_numpy(dtype=float)
        tval = c * v
        for i in range(config.MIN_BAR_INDEX, n - 1):
            if tval[i] < config.MIN_TRADING_VALUE:
                continue
            if not is_breakout(df, i):
                continue
            lab = labels(df, i)
            if not np.isfinite(lab["y_raw"]):
                continue
            rows.append({"code": code, "date": df.index[i], "i": i,
                         "close": c[i], **lab})
    out = pd.DataFrame(rows)
    if len(out):
        out = out.sort_values(["date", "code"]).reset_index(drop=True)
    return out


def add_market_adjusted(events: pd.DataFrame) -> pd.DataFrame:
    """같은 날 이벤트들의 y_raw 중앙값을 빼 시장조정 라벨을 만든다. 스펙 §6."""
    out = events.copy()
    med = out.groupby("date")["y_raw"].transform("median")
    out["y_mkt"] = out["y_raw"] - med
    return out
```

```python
# -*- coding: utf-8 -*-
"""F1 — 기존 프로젝트(stock-screener-kr)의 기술 피처 14개. 스펙 §5.1.

원본: stock-screener-kr/ml/src/collect_training_data.py:calculate_features()
계산식을 그대로 옮기되 기준 시점만 이벤트 봉 i로 통일한다. 재현이 목적이므로
"개선"하지 않는다.
"""
from __future__ import annotations

import numpy as np
import pandas as pd

from surge_d1 import config

F1_NAMES = [
    "breakout_pct", "volume_surge", "close_strength",
    "base_length", "volatility_contraction", "volume_dry_up",
    "ma200_slope", "pct_above_52w_low",
    "rs_vs_market", "market_return",
    "ma20_deviation", "liquidity",
    "days_since_ath", "atr_ratio",
]


def calculate_atr(df: pd.DataFrame, period: int = 14) -> pd.Series:
    """Wilder ATR."""
    h, l, c = df["High"], df["Low"], df["Close"]
    pc = c.shift(1)
    tr = pd.concat([h - l, (h - pc).abs(), (l - pc).abs()], axis=1).max(axis=1)
    return tr.rolling(period).mean()


def _market_return_20d(market: pd.DataFrame, date, i_fallback: int) -> float:
    """시장 20일 누적 수익률(%). market은 index=Date, 컬럼 ret(일간 수익률)."""
    if market is None or date not in market.index:
        return 0.0
    loc = market.index.get_loc(date)
    if isinstance(loc, slice) or loc < 20:
        return 0.0
    window = market["ret"].iloc[loc - 19:loc + 1].fillna(0.0)
    return float((np.prod(1.0 + window.to_numpy()) - 1.0) * 100)


def compute_f1(df: pd.DataFrame, i: int, market: pd.DataFrame) -> dict:
    """봉 i의 F1 피처 14개."""
    close, high, low, vol = df["Close"], df["High"], df["Low"], df["Volume"]
    price = float(close.iloc[i])
    P = config.HIGH_52W_PERIOD

    prior_high = float(close.iloc[max(0, i - P):i].max())
    breakout_pct = (price - prior_high) / prior_high * 100 if prior_high > 0 else 0.0

    vol50 = float(vol.iloc[max(0, i - 50):i].mean())
    volume_surge = float(vol.iloc[i]) / vol50 if vol50 > 0 else 0.0

    rng = float(high.iloc[i] - low.iloc[i])
    close_strength = (price - float(low.iloc[i])) / rng if rng > 0 else 0.5

    # 직전 52주 고가가 나온 위치 이후 경과일
    seg = close.iloc[max(0, i - P):i]
    base_length = int(i - (seg.index.get_loc(seg.idxmax()) + max(0, i - P))) if len(seg) else 0

    r5 = (float(high.iloc[i - 4:i + 1].max()) - float(low.iloc[i - 4:i + 1].min())) / price
    r20 = (float(high.iloc[i - 19:i + 1].max()) - float(low.iloc[i - 19:i + 1].min())) / price
    volatility_contraction = r5 / r20 if r20 > 0 else 1.0

    vol10 = float(vol.iloc[i - 9:i + 1].mean())
    volume_dry_up = vol10 / vol50 if vol50 > 0 else 1.0

    ma200_now = float(close.iloc[i - 199:i + 1].mean())
    ma200_prev = float(close.iloc[i - 219:i - 19].mean())
    ma200_slope = (ma200_now - ma200_prev) / ma200_prev if ma200_prev > 0 else 0.0

    low_52w = float(close.iloc[max(0, i - P):i + 1].min())
    pct_above_52w_low = (price - low_52w) / low_52w * 100 if low_52w > 0 else 0.0

    stock_ret_20d = (price / float(close.iloc[i - 20]) - 1) * 100
    market_return = _market_return_20d(market, df.index[i], i)
    rs_vs_market = stock_ret_20d - market_return

    ma20 = float(close.iloc[i - 19:i + 1].mean())
    ma20_deviation = (price / ma20 - 1) if ma20 > 0 else 0.0

    liquidity = float((close * vol).iloc[i - 4:i + 1].mean())

    ath_seg = close.iloc[:i]
    days_since_ath = int(i - ath_seg.index.get_loc(ath_seg.idxmax())) if len(ath_seg) else 0

    atr = calculate_atr(df, 14)
    atr_val = float(atr.iloc[i]) if np.isfinite(atr.iloc[i]) else 0.0
    atr_ratio = atr_val / price if price > 0 else 0.0

    return {
        "breakout_pct": breakout_pct,
        "volume_surge": volume_surge,
        "close_strength": close_strength,
        "base_length": float(base_length),
        "volatility_contraction": volatility_contraction,
        "volume_dry_up": volume_dry_up,
        "ma200_slope": ma200_slope,
        "pct_above_52w_low": pct_above_52w_low,
        "rs_vs_market": rs_vs_market,
        "market_return": market_return,
        "ma20_deviation": ma20_deviation,
        "liquidity": liquidity,
        "days_since_ath": float(days_since_ath),
        "atr_ratio": atr_ratio,
    }
```

```python
from __future__ import annotations

import numpy as np
import pandas as pd

F2_NAMES = [
    "avg_gap_20d", "gap_vol_20d", "gap_open", "day_change",
    "intraday_range", "upper_shadow", "market_gap_prev", "dow",
    "aligned", "prior_surge_5d",
]

SURGE_LEVEL = 0.12   # '급등' 기준 (모의투자대회 backtest.py와 동일)


def compute_f2(df: pd.DataFrame, i: int, market: pd.DataFrame) -> dict:
    """봉 i의 F2 피처 10개."""
    o = df["Open"].astype(float)
    h = df["High"].astype(float)
    l = df["Low"].astype(float)
    c = df["Close"].astype(float)

    prev_close = float(c.iloc[i - 1])
    price = float(c.iloc[i])

    # 종목별 갭 성향: 최근 20봉의 (Open[t]/Close[t-1] - 1)
    gaps = (o.iloc[i - 19:i + 1].to_numpy() /
            c.iloc[i - 20:i].to_numpy()) - 1.0
    gaps = gaps[np.isfinite(gaps)]
    avg_gap_20d = float(gaps.mean()) if gaps.size else 0.0
    gap_vol_20d = float(gaps.std()) if gaps.size > 1 else 0.0

    gap_open = (float(o.iloc[i]) - prev_close) / prev_close if prev_close > 0 else 0.0
    day_change = (price - prev_close) / prev_close if prev_close > 0 else 0.0

    rng = float(h.iloc[i] - l.iloc[i])
    intraday_range = rng / prev_close if prev_close > 0 else 0.0
    upper_shadow = (float(h.iloc[i]) - price) / rng if rng > 0 else 0.0

    date = df.index[i]
    prev_date = df.index[i - 1]
    market_gap_prev = 0.0
    if market is not None and prev_date in market.index:
        val = market.loc[prev_date, "gap"]
        market_gap_prev = float(val) if np.isfinite(val) else 0.0

    ma20 = float(c.iloc[i - 19:i + 1].mean())
    ma60 = float(c.iloc[i - 59:i + 1].mean())
    aligned = 1.0 if (price > ma20 > ma60) else 0.0

    prior = c.iloc[i - 5:i].to_numpy() / c.iloc[i - 6:i - 1].to_numpy() - 1.0
    prior_surge_5d = 1.0 if np.any(prior >= SURGE_LEVEL) else 0.0

        "avg_gap_20d": avg_gap_20d,
        "gap_vol_20d": gap_vol_20d,
        "gap_open": gap_open,
        "day_change": day_change,
        "intraday_range": intraday_range,
        "upper_shadow": upper_shadow,
        "market_gap_prev": market_gap_prev,
```

```python
# -*- coding: utf-8 -*-
"""F3 — Microsoft qlib Alpha158 계열 피처 20개. 스펙 §5.3.

원 정의: https://github.com/microsoft/qlib — qlib/contrib/data/loader.py 의
Alpha158DL.get_feature_config(). KBAR 9종은 원식 그대로, 롤링은 d=20 창의
11종을 옮겼다. 0으로 나누는 경우 qlib과 동일하게 분모에 1e-12를 더한다.
"""
from __future__ import annotations

import numpy as np
import pandas as pd

EPS = 1e-12
WINDOW = 20

KBAR_NAMES = ["KMID", "KLEN", "KMID2", "KUP", "KUP2",
              "KLOW", "KLOW2", "KSFT", "KSFT2"]
ROLL_NAMES = [f"{n}_{WINDOW}" for n in
              ["RSV", "CORR", "CORD", "CNTP", "SUMP",
               "WVMA", "VSTD", "BETA", "RSQR", "IMAX", "QTLU"]]
F3_NAMES = KBAR_NAMES + ROLL_NAMES


def _kbar(o: float, h: float, l: float, c: float) -> dict:
    """qlib Alpha158 KBAR 9종."""
    hl = (h - l) + EPS
    return {
        "KMID": (c - o) / (o + EPS),
        "KLEN": (h - l) / (o + EPS),
        "KMID2": (c - o) / hl,
        "KUP": (h - max(o, c)) / (o + EPS),
        "KUP2": (h - max(o, c)) / hl,
        "KLOW": (min(o, c) - l) / (o + EPS),
        "KLOW2": (min(o, c) - l) / hl,
        "KSFT": (2 * c - h - l) / (o + EPS),
        "KSFT2": (2 * c - h - l) / hl,
    }


def _safe(x: float) -> float:
    return float(x) if np.isfinite(x) else 0.0


def compute_f3(df: pd.DataFrame, i: int) -> dict:
    """봉 i의 F3 피처 20개 (KBAR 9 + 롤링 11, 창 20)."""
    o = float(df["Open"].iloc[i])
    h = float(df["High"].iloc[i])
    l = float(df["Low"].iloc[i])
    c = float(df["Close"].iloc[i])

    out = {k: _safe(v) for k, v in _kbar(o, h, l, c).items()}

    d = WINDOW
    lo = max(0, i - d + 1)
    close = df["Close"].iloc[lo:i + 1].to_numpy(dtype=float)
    high = df["High"].iloc[lo:i + 1].to_numpy(dtype=float)
    low = df["Low"].iloc[lo:i + 1].to_numpy(dtype=float)
    vol = df["Volume"].iloc[lo:i + 1].to_numpy(dtype=float)
    n = len(close)

    # RSV: 창 내 저가~고가 사이 종가 위치
    rsv = (c - low.min()) / ((high.max() - low.min()) + EPS)

    logv = np.log1p(np.maximum(vol, 0.0))
    corr = np.corrcoef(close, logv)[0, 1] if n > 1 and close.std() > 0 and logv.std() > 0 else 0.0

    dc = np.diff(close) / (close[:-1] + EPS)
    dv = np.diff(logv)
    cord = (np.corrcoef(dc, dv)[0, 1]
            if len(dc) > 1 and dc.std() > 0 and dv.std() > 0 else 0.0)

    cntp = float((dc > 0).mean()) if len(dc) else 0.0
    gains = np.maximum(dc, 0.0).sum()
    total = np.abs(dc).sum()
    sump = gains / (total + EPS)

    wvma = (np.std(dc * vol[1:]) / (np.mean(np.abs(dc * vol[1:])) + EPS)
            if len(dc) else 0.0)
    vstd = vol.std() / (vol.mean() + EPS)

    x = np.arange(n, dtype=float)
    if n > 1 and close.std() > 0:
        slope, intercept = np.polyfit(x, close, 1)
        pred = slope * x + intercept
        ss_res = float(((close - pred) ** 2).sum())
        ss_tot = float(((close - close.mean()) ** 2).sum())
        rsqr = 1.0 - ss_res / (ss_tot + EPS)
    else:
        slope, rsqr = 0.0, 0.0

    imax = float(n - 1 - int(np.argmax(high))) / d
    qtlu = float(np.quantile(close, 0.8)) / (c + EPS)

    out.update({
        f"RSV_{d}": _safe(rsv),
        f"CORR_{d}": _safe(corr),
        f"CORD_{d}": _safe(cord),
        f"CNTP_{d}": _safe(cntp),
        f"SUMP_{d}": _safe(sump),
        f"WVMA_{d}": _safe(wvma),
        f"VSTD_{d}": _safe(vstd),
        f"BETA_{d}": _safe(slope / (c + EPS)),
        f"RSQR_{d}": _safe(rsqr),
        f"IMAX_{d}": _safe(imax),
        f"QTLU_{d}": _safe(qtlu),
    })
    return out
```

```python
# -*- coding: utf-8 -*-
"""F4 — 재료 피처 5개. 스펙 §5.4.

핵심 제약: type_prior와 TF-IDF 인덱스는 **학습 구간 데이터만으로** 만든다.
전역 상수(CATALYST_TYPE_PRIOR)를 쓰면 미래정보 누수이므로 쓰지 않는다.
원 아이디어: 모의투자대회/catalyst_engine/baserate.py (char n-gram TF-IDF).
"""
from __future__ import annotations

import numpy as np
import pandas as pd

F4_NAMES = ["type_prior", "analog_mean_d1", "analog_win_rate",
            "analog_n", "reason_len"]

NEUTRAL = {"type_prior": 0.0, "analog_mean_d1": 0.0,
           "analog_win_rate": 0.5, "analog_n": 0.0, "reason_len": 0.0}


class CatalystIndex:
    """학습 구간 재료 데이터로만 만든 prior + 유사사례 검색기."""

    def __init__(self) -> None:
        self._fitted = False
        self._type_mean: dict[str, float] = {}
        self._global_mean = 0.0
        self._vec = None
        self._mat = None
        self._d1 = np.array([])

    def fit(self, rows: pd.DataFrame) -> "CatalystIndex":
        """rows: 컬럼 ctype, reason, d1 (학습 구간만 넘길 것)."""
        from sklearn.feature_extraction.text import TfidfVectorizer

        r = rows.dropna(subset=["d1"]).copy()
        r["reason"] = r["reason"].fillna("").astype(str)
        r = r[r["reason"].str.len() > 0]

        self._global_mean = float(r["d1"].mean()) if len(r) else 0.0
        self._type_mean = r.groupby("ctype")["d1"].mean().to_dict()
        self._d1 = r["d1"].to_numpy(dtype=float)

        if len(r) >= 2:
            self._vec = TfidfVectorizer(analyzer="char_wb", ngram_range=(2, 4),
                                        min_df=1, sublinear_tf=True)
            self._mat = self._vec.fit_transform(r["reason"].tolist())
        self._fitted = True
        return self

    def transform(self, ctype, reason, k: int = 10) -> dict:
        """재료 유형·사유 → F4 피처 5개. 재료가 없으면 중립값."""
        if not self._fitted:
            raise RuntimeError("CatalystIndex.fit()을 먼저 호출해야 한다")
        if not reason or (isinstance(reason, float) and np.isnan(reason)):
            return dict(NEUTRAL)

        reason = str(reason)
        out = dict(NEUTRAL)

                out["analog_win_rate"] = float((vals > 0).mean())

        subset=["date", "code"], keep="first")

    out["has_catalyst"] = out["reason"].notna().astype(int)
    return out
```

```python
# -*- coding: utf-8 -*-
"""검증 프로토콜: walk-forward + embargo, 지표, 통계검정. 스펙 §7.2, §7.5."""
from __future__ import annotations

import numpy as np
import pandas as pd
from scipy import stats

from surge_d1 import config


def walk_forward_folds(dates: pd.Series, test_years: list[int],
                       embargo_days: int = config.EMBARGO_DAYS
                       ) -> list[tuple[np.ndarray, np.ndarray]]:
    """expanding window walk-forward. 반환: [(train_mask, test_mask), ...]

    학습은 검정연도 시작 이전 전체에서 embargo_days(달력일) 만큼을 더 잘라낸다.
    """
    d = pd.to_datetime(pd.Series(dates).reset_index(drop=True))
    folds = []
    for y in sorted(test_years):
        start = pd.Timestamp(f"{y}-01-01")
        end = pd.Timestamp(f"{y}-12-31 23:59:59")
        test = (d >= start) & (d <= end)
        cutoff = start - pd.Timedelta(days=embargo_days)
        train = d < cutoff
        if train.sum() == 0 or test.sum() == 0:
            continue
        folds.append((train.to_numpy(), test.to_numpy()))
    return folds


def spearman_ic(y_true, y_pred, dates) -> tuple[float, float]:
    """일별 Spearman 순위상관의 평균과 t값(=IR 근사)."""
    df = pd.DataFrame({"y": np.asarray(y_true, dtype=float),
                       "p": np.asarray(y_pred, dtype=float),
                       "d": pd.to_datetime(pd.Series(dates).values)})
    ics = []
    for _, g in df.groupby("d"):
        if len(g) < 3 or g["p"].nunique() < 2 or g["y"].nunique() < 2:
            continue
        ic = stats.spearmanr(g["y"], g["p"]).statistic
        if np.isfinite(ic):
            ics.append(float(ic))
    if not ics:
        return 0.0, 0.0
    arr = np.array(ics)
    se = arr.std(ddof=1) / np.sqrt(len(arr)) if len(arr) > 1 else np.nan
    t = float(arr.mean() / se) if se and np.isfinite(se) and se > 0 else 0.0
    return float(arr.mean()), t


def top_decile_return(y_true, y_pred, dates, q: float = 0.1) -> float:
    """일별로 예측 상위 q 비율을 뽑아 실제 수익률 평균을 낸다."""
    df = pd.DataFrame({"y": np.asarray(y_true, dtype=float),
                       "p": np.asarray(y_pred, dtype=float),
                       "d": pd.to_datetime(pd.Series(dates).values)})
    picked = []
    for _, g in df.groupby("d"):
        k = max(1, int(round(len(g) * q)))
        picked.append(g.nlargest(k, "p")["y"])
    if not picked:
        return float("nan")
    return float(pd.concat(picked).mean())


def bootstrap_ci(values, n: int = 1000, seed: int = config.RANDOM_SEED,
                 alpha: float = 0.05) -> tuple[float, float]:
    """평균의 부트스트랩 신뢰구간."""
    v = np.asarray(values, dtype=float)
    v = v[np.isfinite(v)]
    if v.size == 0:
        return float("nan"), float("nan")
    rng = np.random.default_rng(seed)
    means = rng.choice(v, size=(n, v.size), replace=True).mean(axis=1)
    return (float(np.quantile(means, alpha / 2)),
            float(np.quantile(means, 1 - alpha / 2)))


def permutation_pvalue(metric_fn, y, pred_a, pred_b, dates,
                       n: int = 1000, seed: int = config.RANDOM_SEED) -> float:
    """H0: pred_b가 pred_a보다 낫지 않다.

    관측 차이(metric(b) - metric(a))가, b의 예측을 무작위 섞었을 때의 분포에서
    얼마나 극단적인지로 p값을 낸다.
    """
    y = np.asarray(y, dtype=float)
    a = np.asarray(pred_a, dtype=float)
    b = np.asarray(pred_b, dtype=float)
    observed = metric_fn(y, b, dates) - metric_fn(y, a, dates)
    rng = np.random.default_rng(seed)
    cnt = 0
    for _ in range(n):
        shuffled = rng.permutation(b)
        diff = metric_fn(y, shuffled, dates) - metric_fn(y, a, dates)
        if diff >= observed:
            cnt += 1
    return (cnt + 1) / (n + 1)


def benjamini_hochberg(pvals: list[float]) -> list[float]:
    """BH 보정 p값. 입력 순서를 유지해 반환한다."""
    p = np.asarray(pvals, dtype=float)
    m = p.size
    order = np.argsort(p)
    ranked = p[order]
    adj = ranked * m / (np.arange(m) + 1)
    adj = np.minimum.accumulate(adj[::-1])[::-1]
    adj = np.clip(adj, 0.0, 1.0)
    out = np.empty(m, dtype=float)
    out[order] = adj
    return out.tolist()
```

```python
                  subsample=0.8, reg_alpha=1.0, reg_lambda=5.0,
                  random_state=config.RANDOM_SEED, verbose=-1, n_jobs=-1)
CLF_PARAMS = dict(n_estimators=200, max_depth=5, learning_rate=0.05,
                  subsample=0.8, colsample_bytree=0.8,
                  random_state=config.RANDOM_SEED, verbose=-1, n_jobs=-1)

TEST_YEARS = [2023, 2024, 2025, 2026]


def fit_predict(train_df: pd.DataFrame, test_df: pd.DataFrame,
                feats: list[str], use_weight: bool = True,
                label_reg: str = "y_raw") -> dict:
    """회귀·분류 학습 후 예측. feats가 비면 M0(상수 예측)."""
    n_test = len(test_df)
    if not feats:
        base_ret = float(train_df[label_reg].mean())
        base_p = float(train_df["y_cls"].mean())
        return {"pred_reg": np.full(n_test, base_ret),
                "prob_raw": np.full(n_test, base_p),
                "prob_cal": np.full(n_test, base_p)}

    from lightgbm import LGBMClassifier, LGBMRegressor

    Xtr = train_df[feats].to_numpy(dtype=float)
    Xte = test_df[feats].to_numpy(dtype=float)
    w = train_df["weight"].to_numpy(dtype=float) if use_weight else None

    reg = LGBMRegressor(**REG_PARAMS)
    reg.fit(Xtr, train_df[label_reg].to_numpy(dtype=float), sample_weight=w)
    pred_reg = reg.predict(Xte)

    clf = LGBMClassifier(**CLF_PARAMS)
    ytr = train_df["y_cls"].to_numpy(dtype=int)
    clf.fit(Xtr, ytr, sample_weight=w)
    prob_raw = clf.predict_proba(Xte)[:, 1]

    # 확률 보정: 학습 구간 내부 예측으로 isotonic을 적합 (스펙 §7.4)
    prob_tr = clf.predict_proba(Xtr)[:, 1]
    iso = IsotonicRegression(out_of_bounds="clip", y_min=0.0, y_max=1.0)
    iso.fit(prob_tr, ytr)
    prob_cal = np.clip(iso.predict(prob_raw), 0.0, 1.0)

    return {"pred_reg": pred_reg, "prob_raw": prob_raw, "prob_cal": prob_cal}


def evaluate(test_df: pd.DataFrame, out: dict,
             label_reg: str = "y_raw") -> dict:
    """검정 구간 지표. 스펙 §7.5."""
    y = test_df[label_reg].to_numpy(dtype=float)
    ycls = test_df["y_cls"].to_numpy(dtype=int)
    dates = test_df["date"]

    if len(np.unique(ycls)) < 2:
        auc = float("nan")
    else:
        auc = float(roc_auc_score(ycls, out["prob_cal"]))
    brier = float(brier_score_loss(ycls, out["prob_cal"]))
    ic, ic_t = validate.spearman_ic(y, out["pred_reg"], dates)
    top = validate.top_decile_return(y, out["pred_reg"], dates)
    return {"auc": auc, "brier": brier, "ic": ic, "ic_t": ic_t,
            "top_decile": top, "n": int(len(test_df))}


def _build_f4(train_df: pd.DataFrame, test_df: pd.DataFrame
              ) -> tuple[pd.DataFrame, pd.DataFrame]:
    """폴드 학습 구간만으로 CatalystIndex를 적합해 F4를 만든다. 스펙 §5.4."""
    idx = features_f4.CatalystIndex().fit(
        train_df.loc[train_df["has_catalyst"] == 1, ["ctype", "reason", "d1"]])

    def apply(df):
        rows = [idx.transform(r.ctype, r.reason)
                for r in df.itertuples(index=False)]
        return pd.concat([df.reset_index(drop=True),
                          pd.DataFrame(rows)], axis=1)

    return apply(train_df), apply(test_df)


def run_ablation(df: pd.DataFrame) -> pd.DataFrame:
    """M0~M3는 전체 표본 4폴드, M3prime/M4는 재료 표본 단일 폴드."""
    results = []

    folds = validate.walk_forward_folds(df["date"], TEST_YEARS)
    for fi, (tr_mask, te_mask) in enumerate(folds, 1):
        tr, te = df[tr_mask].reset_index(drop=True), df[te_mask].reset_index(drop=True)
        year = int(te["date"].dt.year.mode()[0])
        for name in ("M0", "M1", "M2", "M3"):
            feats = [] if name == "M0" else dataset.FEATURE_SETS[name]
            out = fit_predict(tr, te, feats)
            results.append({"model": name, "fold": fi, "test_year": year,
                            "sample": "full", **evaluate(te, out)})
            print(f"  {name} fold{fi}({year}) 완료", flush=True)

    # 재료 표본: M3prime vs M4
    cat = df[df["has_catalyst"] == 1].reset_index(drop=True)
    if len(cat) > 200:
        split = cat["date"].quantile(0.6)
        tr = cat[cat["date"] < split].reset_index(drop=True)
        te = cat[cat["date"] >= split].reset_index(drop=True)
        tr4, te4 = _build_f4(tr, te)
        for name, feats in (("M3prime", dataset.FEATURE_SETS["M3"]),
                            ("M4", dataset.FEATURE_SETS["M4"])):
            out = fit_predict(tr4, te4, feats)
            results.append({"model": name, "fold": 1,
                            "test_year": int(te["date"].dt.year.min()),
                            "sample": "catalyst", **evaluate(te4, out)})
            print(f"  {name} (재료 표본) 완료", flush=True)

    return pd.DataFrame(results)


def run_leakage_comparison(df: pd.DataFrame) -> pd.DataFrame:
    """같은 데이터를 셔플 CV와 walk-forward로 평가해 편향을 잰다. 스펙 §7.6."""
    feats = dataset.FEATURE_SETS["M1"]
    rows = []

    kf = KFold(n_splits=5, shuffle=True, random_state=config.RANDOM_SEED)
    aucs, ics = [], []
    for tr_i, te_i in kf.split(df):
        tr = df.iloc[tr_i].reset_index(drop=True)
        te = df.iloc[te_i].reset_index(drop=True)
        m = evaluate(te, fit_predict(tr, te, feats))
        aucs.append(m["auc"])
        ics.append(m["ic"])
    rows.append({"method": "shuffle_kfold", "auc": float(np.nanmean(aucs)),
                 "ic": float(np.nanmean(ics))})

    aucs, ics = [], []
    for tr_mask, te_mask in validate.walk_forward_folds(df["date"], TEST_YEARS):
        tr = df[tr_mask].reset_index(drop=True)
        te = df[te_mask].reset_index(drop=True)
        m = evaluate(te, fit_predict(tr, te, feats))
        aucs.append(m["auc"])
        ics.append(m["ic"])
    rows.append({"method": "walk_forward", "auc": float(np.nanmean(aucs)),
                 "ic": float(np.nanmean(ics))})

    out = pd.DataFrame(rows)
    out["auc_bias"] = out["auc"] - out["auc"].iloc[1]
    return out


def main() -> int:
    sys.stdout.reconfigure(encoding="utf-8")
    df = pd.read_parquet(config.DATASET_PATH)
    config.RESULTS_DIR.mkdir(parents=True, exist_ok=True)

    print("=== ablation (M0~M4) ===")
    ab = run_ablation(df)
    ab.to_csv(config.RESULTS_DIR / "ablation.csv", index=False, encoding="utf-8-sig")
    print(ab.groupby("model")[["auc", "ic", "top_decile"]].mean().to_string())

    print("\n=== 누수 편향 비교 ===")
    lk = run_leakage_comparison(df)
    lk.to_csv(config.RESULTS_DIR / "leakage.csv", index=False, encoding="utf-8-sig")
    print(lk.to_string(index=False))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# -*- coding: utf-8 -*-
"""오늘 52주 신고가를 돌파한 종목의 익일 상승확률·기대수익률을 산출한다.

학습은 저장된 데이터셋 전체로 하고, 예측은 최신 봉 후보에 대해 한다.
익일 데이터가 없으므로 라벨 없이 피처만 계산한다.

사용:
    python predict.py                  # 캐시 최신일 기준
    python predict.py --asof 20260828  # 특정일 기준
"""
from __future__ import annotations

import argparse
import sys

import numpy as np
import pandas as pd

from surge_d1 import (config, dataset, features_f1, features_f2,
                      features_f3, models, sources)


def _is_breakout_last(df: pd.DataFrame, i: int) -> bool:
    """익일 존재 조건을 뺀 돌파 판정 (예측 전용)."""
    if i < config.MIN_BAR_INDEX or i >= len(df):
        return False
    c = float(df["Close"].iloc[i])
    v = float(df["Volume"].iloc[i])
    if not np.isfinite(c) or not np.isfinite(v) or c <= 0:
        return False
    if not (c > float(df["Close"].iloc[i - config.HIGH_52W_PERIOD:i].max())):
        return False
    vol_avg = float(df["Volume"].iloc[i - 50:i].mean())
    if not (vol_avg > 0 and v >= config.VOLUME_SURGE_MULT * vol_avg):
        return False
    return bool(c * v >= config.MIN_TRADING_VALUE)


def find_today_events(ohlcv: dict, market: pd.DataFrame,
                      asof: pd.Timestamp) -> pd.DataFrame:
    """asof 날짜에 돌파한 종목과 그 피처(M3)를 모은다."""
    rows = []
    for code, df in ohlcv.items():
        if asof not in df.index:
            continue
        i = int(df.index.get_loc(asof))
        if not _is_breakout_last(df, i):
            continue
        feat = {}
        feat.update(features_f1.compute_f1(df, i, market))
        feat.update(features_f2.compute_f2(df, i, market))
        feat.update(features_f3.compute_f3(df, i))
        rows.append({"code": code, "date": asof,
                     "close": float(df["Close"].iloc[i]), **feat})
    return pd.DataFrame(rows)


def main() -> int:
    sys.stdout.reconfigure(encoding="utf-8")
    ap = argparse.ArgumentParser()
    ap.add_argument("--asof", default=None, help="YYYYMMDD (기본: 캐시 최신일)")
    args = ap.parse_args()

    ohlcv = sources.load_ohlcv()
    market = sources.build_market_index(ohlcv)
    asof = (pd.Timestamp(args.asof) if args.asof
            else max(df.index[-1] for df in ohlcv.values()))
    print(f"기준일: {asof.date()}")

    cand = find_today_events(ohlcv, market, asof)
    if cand.empty:
        print("해당일 돌파 종목 없음")
        return 0
    print(f"돌파 종목 {len(cand)}개")

    train = pd.read_parquet(config.DATASET_PATH)
    train = train[train["date"] < asof]
    feats = dataset.FEATURE_SETS["M3"]
    out = models.fit_predict(train, cand, feats)

    cand["상승확률"] = out["prob_cal"]
    cand["기대수익률"] = out["pred_reg"]
    res = (cand[["code", "date", "close", "상승확률", "기대수익률"]]
           .sort_values("상승확률", ascending=False).reset_index(drop=True))
    res["상승확률"] = (res["상승확률"] * 100).round(1)
    res["기대수익률"] = (res["기대수익률"] * 100).round(3)

    config.RESULTS_DIR.mkdir(parents=True, exist_ok=True)
    path = config.RESULTS_DIR / f"predict_{asof.strftime('%Y%m%d')}.csv"
    res.to_csv(path, index=False, encoding="utf-8-sig")
    print(res.head(20).to_string(index=False))
    print(f"\n저장: {path}")
    print("주의: 이 수치는 연구용 추정치이며 투자 권유가 아니다.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
